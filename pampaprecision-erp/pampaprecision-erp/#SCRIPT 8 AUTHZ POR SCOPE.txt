-- Requiere: sec.app_user, sec.role, sec.permission,
--           sec.user_role, sec.role_permission, sec.data_access_scope
-- Uso esperado:
--   SET app.user_id = '...uuid...';
--   SET app.tenant_id = '...uuid...';
-- =========================================================

CREATE SCHEMA IF NOT EXISTS authz;

-- Contexto actual tomado de variables de sesión
CREATE OR REPLACE FUNCTION authz.current_user_id()
RETURNS uuid
LANGUAGE sql
STABLE
AS $$
  SELECT NULLIF(current_setting('app.user_id', true), '')::uuid
$$;

CREATE OR REPLACE FUNCTION authz.current_tenant_id()
RETURNS uuid
LANGUAGE sql
STABLE
AS $$
  SELECT NULLIF(current_setting('app.tenant_id', true), '')::uuid
$$;

-- Verifica permiso funcional (erp.read, precision.write, etc.)
CREATE OR REPLACE FUNCTION authz.has_permission(p_permission_code text)
RETURNS boolean
LANGUAGE sql
STABLE
AS $$
  SELECT EXISTS (
    SELECT 1
    FROM sec.user_role ur
    JOIN sec.role_permission rp ON rp.role_id = ur.role_id
    JOIN sec.permission p ON p.id = rp.permission_id
    JOIN sec.app_user u ON u.id = ur.user_id
    WHERE ur.user_id = authz.current_user_id()
      AND u.tenant_id = authz.current_tenant_id()
      AND p.code = p_permission_code
  )
$$;

-- Scope por lote
CREATE OR REPLACE FUNCTION authz.has_scope_plot(p_plot_id uuid)
RETURNS boolean
LANGUAGE sql
STABLE
AS $$
  SELECT EXISTS (
    SELECT 1
    FROM sec.data_access_scope s
    WHERE s.tenant_id = authz.current_tenant_id()
      AND s.user_id = authz.current_user_id()
      AND (
        s.plot_id = p_plot_id
        OR (
          s.plot_id IS NULL
          AND s.establishment_id IS NOT NULL
          AND s.establishment_id = (SELECT establishment_id FROM core.plot WHERE id = p_plot_id)
        )
        OR (
          s.plot_id IS NULL
          AND s.establishment_id IS NULL
          AND s.campaign_id IS NULL
        )
      )
  )
$$;

-- Scope por lote-campaña
CREATE OR REPLACE FUNCTION authz.has_scope_plot_season(p_plot_season_id uuid)
RETURNS boolean
LANGUAGE sql
STABLE
AS $$
  SELECT EXISTS (
    SELECT 1
    FROM core.plot_season ps
    JOIN core.plot p ON p.id = ps.plot_id
    WHERE ps.id = p_plot_season_id
      AND EXISTS (
        SELECT 1
        FROM sec.data_access_scope s
        WHERE s.tenant_id = authz.current_tenant_id()
          AND s.user_id = authz.current_user_id()
          AND (
            s.plot_id = ps.plot_id
            OR (
              s.plot_id IS NULL
              AND s.establishment_id = p.establishment_id
              AND (s.campaign_id IS NULL OR s.campaign_id = ps.campaign_id)
            )
            OR (
              s.plot_id IS NULL
              AND s.establishment_id IS NULL
              AND (s.campaign_id IS NULL OR s.campaign_id = ps.campaign_id)
            )
          )
      )
  )
$$;

-- Falla explícitamente si no cumple permiso
CREATE OR REPLACE FUNCTION authz.assert_permission(p_permission_code text)
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  IF NOT authz.has_permission(p_permission_code) THEN
    RAISE EXCEPTION 'forbidden_permission: %', p_permission_code
      USING ERRCODE = '42501';
  END IF;
END;
$$;

-- Falla explícitamente si no cumple scope de lote-campaña
CREATE OR REPLACE FUNCTION authz.assert_scope_plot_season(p_plot_season_id uuid)
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  IF NOT authz.has_scope_plot_season(p_plot_season_id) THEN
    RAISE EXCEPTION 'forbidden_scope_plot_season: %', p_plot_season_id
      USING ERRCODE = '42501';
  END IF;
END;
$$;

-- Ejemplo de uso en una función de aplicación
CREATE OR REPLACE FUNCTION ops.secure_create_work_order(
  p_plot_season_id uuid,
  p_operation_type varchar,
  p_requested_date date,
  p_planned_area_ha numeric
)
RETURNS uuid
LANGUAGE plpgsql
AS $$
DECLARE
  v_id uuid := gen_random_uuid();
BEGIN
  PERFORM authz.assert_permission('erp.write');
  PERFORM authz.assert_scope_plot_season(p_plot_season_id);

  INSERT INTO ops.work_order (
    id, tenant_id, plot_season_id, operation_type, requested_date, planned_area_ha, status
  )
  VALUES (
    v_id, authz.current_tenant_id(), p_plot_season_id, p_operation_type, p_requested_date, p_planned_area_ha, 'open'
  );

  RETURN v_id;
END;
$$;
