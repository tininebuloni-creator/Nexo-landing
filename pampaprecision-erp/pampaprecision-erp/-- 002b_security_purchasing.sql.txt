-- Tablas requeridas por autorizacion por scope y auditoria automatica.

CREATE TABLE IF NOT EXISTS inv.purchase_order (
  id uuid PRIMARY KEY,
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  supplier_party_id uuid NOT NULL,
  po_number varchar(40) NOT NULL,
  po_date date NOT NULL,
  status varchar(20) NOT NULL DEFAULT 'open',
  currency_id uuid NOT NULL REFERENCES ref.currency(id),
  UNIQUE (tenant_id, po_number)
);

CREATE TABLE IF NOT EXISTS inv.purchase_order_line (
  id uuid PRIMARY KEY,
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  purchase_order_id uuid NOT NULL REFERENCES inv.purchase_order(id),
  input_id uuid NOT NULL REFERENCES agr.input(id),
  qty numeric(14,4) NOT NULL,
  uom_id uuid NOT NULL REFERENCES ref.uom(id),
  unit_price numeric(16,4) NOT NULL
);

CREATE TABLE IF NOT EXISTS inv.goods_receipt (
  id uuid PRIMARY KEY,
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  purchase_order_id uuid NOT NULL REFERENCES inv.purchase_order(id),
  receipt_number varchar(40) NOT NULL,
  receipt_date date NOT NULL,
  UNIQUE (tenant_id, receipt_number)
);

CREATE TABLE IF NOT EXISTS inv.goods_receipt_line (
  id uuid PRIMARY KEY,
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  goods_receipt_id uuid NOT NULL REFERENCES inv.goods_receipt(id),
  purchase_order_line_id uuid NOT NULL REFERENCES inv.purchase_order_line(id),
  received_qty numeric(14,4) NOT NULL
);

CREATE TABLE IF NOT EXISTS inv.supplier_invoice (
  id uuid PRIMARY KEY,
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  purchase_order_id uuid REFERENCES inv.purchase_order(id),
  invoice_number varchar(40) NOT NULL,
  invoice_date date NOT NULL,
  total_amount numeric(16,2) NOT NULL,
  currency_id uuid NOT NULL REFERENCES ref.currency(id),
  UNIQUE (tenant_id, invoice_number)
);

CREATE TABLE IF NOT EXISTS sec.data_access_scope (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  user_id uuid NOT NULL REFERENCES sec.app_user(id),
  campaign_id uuid REFERENCES core.campaign(id),
  establishment_id uuid REFERENCES core.establishment(id),
  plot_id uuid REFERENCES core.plot(id)
);
