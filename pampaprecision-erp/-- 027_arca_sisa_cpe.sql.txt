-- Integracion ARCA: consultas SISA y Carta de Porte Electronica.
CREATE TABLE IF NOT EXISTS fin.arca_sisa_check (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  subject_tax_id varchar(30) NOT NULL,
  subject_name text,
  sisa_status varchar(30),
  registration_status varchar(40),
  result_payload jsonb NOT NULL DEFAULT '{}'::jsonb,
  status varchar(20) NOT NULL DEFAULT 'pending',
  error_message text,
  checked_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS fin.arca_cpe (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  lpg_id uuid REFERENCES fin.lpg(id),
  cpe_number varchar(40),
  authorization_code varchar(60),
  operation_type varchar(30) NOT NULL DEFAULT 'request',
  status varchar(20) NOT NULL DEFAULT 'draft',
  issuer_tax_id varchar(30) NOT NULL,
  producer_tax_id varchar(30) NOT NULL,
  origin_province text NOT NULL,
  origin_locality text,
  destination text NOT NULL,
  grain_code varchar(30) NOT NULL,
  grain_name text NOT NULL,
  campaign varchar(20),
  gross_weight_kg numeric(16,3) NOT NULL,
  vehicle_plate varchar(20),
  trailer_plate varchar(20),
  driver_tax_id varchar(30),
  driver_name text,
  payload jsonb NOT NULL DEFAULT '{}'::jsonb,
  response_payload jsonb NOT NULL DEFAULT '{}'::jsonb,
  error_message text,
  requested_at timestamptz NOT NULL DEFAULT now(),
  authorized_at timestamptz
);

CREATE INDEX IF NOT EXISTS idx_arca_sisa_tenant_tax ON fin.arca_sisa_check (tenant_id, subject_tax_id, checked_at DESC);
CREATE INDEX IF NOT EXISTS idx_arca_cpe_tenant_status ON fin.arca_cpe (tenant_id, status, requested_at DESC);
