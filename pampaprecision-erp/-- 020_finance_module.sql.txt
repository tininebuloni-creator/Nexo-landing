-- Modulo financiero: ventas, fondos y flujo de caja.

CREATE TABLE IF NOT EXISTS fin.cash_account (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  name text NOT NULL,
  account_type varchar(20) NOT NULL DEFAULT 'bank',
  currency_id uuid NOT NULL REFERENCES ref.currency(id),
  opening_balance numeric(16,2) NOT NULL DEFAULT 0,
  active boolean NOT NULL DEFAULT true,
  UNIQUE (tenant_id, name)
);

CREATE TABLE IF NOT EXISTS fin.sale (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  sale_number varchar(40) NOT NULL,
  sale_date date NOT NULL DEFAULT CURRENT_DATE,
  customer_name text NOT NULL,
  customer_tax_id varchar(30),
  currency_id uuid NOT NULL REFERENCES ref.currency(id),
  subtotal numeric(16,2) NOT NULL DEFAULT 0,
  tax_amount numeric(16,2) NOT NULL DEFAULT 0,
  total_amount numeric(16,2) NOT NULL DEFAULT 0,
  due_date date,
  status varchar(20) NOT NULL DEFAULT 'issued',
  payment_status varchar(20) NOT NULL DEFAULT 'pending',
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (tenant_id, sale_number),
  CHECK (subtotal >= 0 AND tax_amount >= 0 AND total_amount >= 0)
);

CREATE TABLE IF NOT EXISTS fin.sale_line (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  sale_id uuid NOT NULL REFERENCES fin.sale(id) ON DELETE CASCADE,
  description text NOT NULL,
  quantity numeric(14,3) NOT NULL,
  unit_price numeric(16,2) NOT NULL,
  tax_rate numeric(6,3) NOT NULL DEFAULT 0,
  line_total numeric(16,2) NOT NULL,
  CHECK (quantity > 0 AND unit_price >= 0 AND tax_rate >= 0)
);

CREATE TABLE IF NOT EXISTS fin.cash_movement (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  cash_account_id uuid NOT NULL REFERENCES fin.cash_account(id),
  movement_date date NOT NULL DEFAULT CURRENT_DATE,
  movement_type varchar(10) NOT NULL,
  source_type varchar(30) NOT NULL DEFAULT 'manual',
  source_id uuid,
  description text NOT NULL,
  amount_usd numeric(16,2) NOT NULL,
  status varchar(20) NOT NULL DEFAULT 'confirmed',
  created_at timestamptz NOT NULL DEFAULT now(),
  CHECK (movement_type IN ('in', 'out')),
  CHECK (amount_usd > 0)
);

CREATE INDEX IF NOT EXISTS idx_sale_tenant_date ON fin.sale (tenant_id, sale_date DESC);
CREATE INDEX IF NOT EXISTS idx_cash_movement_tenant_date ON fin.cash_movement (tenant_id, movement_date DESC);
CREATE INDEX IF NOT EXISTS idx_cash_movement_source ON fin.cash_movement (source_type, source_id);
