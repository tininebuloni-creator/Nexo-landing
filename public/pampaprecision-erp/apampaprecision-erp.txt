CREATE EXTENSION IF NOT EXISTS postgis;

CREATE SCHEMA IF NOT EXISTS core;

-- =========================
-- REFERENCIAS
-- =========================
CREATE TABLE ref.currency (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code varchar(3) NOT NULL UNIQUE,
  name text NOT NULL
);

CREATE TABLE ref.uom (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code varchar(30) NOT NULL UNIQUE,
  name text NOT NULL,
  dimension varchar(30) NOT NULL
);

CREATE TABLE ref.uom_conversion (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  from_uom_id uuid NOT NULL REFERENCES ref.uom(id),
  to_uom_id uuid NOT NULL REFERENCES ref.uom(id),
  factor numeric(20,8) NOT NULL,
  UNIQUE (from_uom_id, to_uom_id)
);

-- =========================
-- CORE
-- =========================
CREATE TABLE core.tenant (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE core.campaign (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  code varchar(30) NOT NULL,
  name text NOT NULL,
  crop_year_start date NOT NULL,
  crop_year_end date NOT NULL,
  currency_id uuid NOT NULL REFERENCES ref.currency(id),
  is_open boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (tenant_id, code),
  CHECK (crop_year_end > crop_year_start)
);

CREATE TABLE core.establishment (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  code varchar(30) NOT NULL,
  name text NOT NULL,
  country_code varchar(2) NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (tenant_id, code)
);

CREATE TABLE core.plot (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  establishment_id uuid NOT NULL REFERENCES core.establishment(id),
  code varchar(30) NOT NULL,
  name text NOT NULL,
  area_ha_declared numeric(12,3) NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (establishment_id, code)
);

CREATE TABLE core.plot_season (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  campaign_id uuid NOT NULL REFERENCES core.campaign(id),
  plot_id uuid NOT NULL REFERENCES core.plot(id),
  planned_crop_id uuid,
  actual_crop_id uuid,
  status varchar(20) NOT NULL DEFAULT 'planned',
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (campaign_id, plot_id)
);

-- =========================
-- AGRONOMIA
-- =========================
CREATE TABLE agr.crop (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  code varchar(30) NOT NULL,
  name text NOT NULL,
  UNIQUE (tenant_id, code)
);

CREATE TABLE agr.variety (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  crop_id uuid NOT NULL REFERENCES agr.crop(id),
  code varchar(30) NOT NULL,
  name text NOT NULL,
  UNIQUE (tenant_id, code)
);

CREATE TABLE agr.input (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  type varchar(20) NOT NULL,
  commercial_name text NOT NULL,
  uom_id uuid NOT NULL REFERENCES ref.uom(id),
  UNIQUE (tenant_id, commercial_name),
  CHECK (type IN ('seed','fertilizer','phytosanitary','fuel','other'))
);

CREATE TABLE agr.active_ingredient (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  name text NOT NULL,
  UNIQUE (tenant_id, name)
);

CREATE TABLE agr.input_active_ingredient (
  input_id uuid NOT NULL REFERENCES agr.input(id),
  active_ingredient_id uuid NOT NULL REFERENCES agr.active_ingredient(id),
  concentration numeric(10,4),
  PRIMARY KEY (input_id, active_ingredient_id)
);

CREATE TABLE agr.input_regulatory_profile (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  input_id uuid NOT NULL REFERENCES agr.input(id),
  country_code varchar(2) NOT NULL,
  min_dose numeric(12,4),
  max_dose numeric(12,4),
  withholding_days integer,
  reentry_hours integer,
  UNIQUE (input_id, country_code)
);

-- =========================
-- PLANIFICACION
-- =========================
CREATE TABLE agr.technical_budget_line (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  plot_season_id uuid NOT NULL REFERENCES core.plot_season(id),
  concept varchar(50) NOT NULL,
  input_id uuid REFERENCES agr.input(id),
  qty_per_ha numeric(14,4) NOT NULL,
  uom_id uuid NOT NULL REFERENCES ref.uom(id),
  unit_cost numeric(16,4) NOT NULL,
  total_cost_per_ha numeric(16,4) GENERATED ALWAYS AS (qty_per_ha * unit_cost) STORED
);

-- =========================
-- OPERACIONES

CREATE TABLE ops.work_order (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  plot_season_id uuid NOT NULL REFERENCES core.plot_season(id),
  operation_type varchar(30) NOT NULL,
  requested_date date NOT NULL,
  planned_area_ha numeric(12,3) NOT NULL,
  status varchar(20) NOT NULL DEFAULT 'open',
  contractor_party_id uuid,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE ops.work_execution (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  work_order_id uuid NOT NULL REFERENCES ops.work_order(id),
  executed_at timestamptz NOT NULL,
  area_ha numeric(12,3) NOT NULL,
  machine_hours numeric(12,2) NOT NULL DEFAULT 0,
  labor_hours numeric(12,2) NOT NULL DEFAULT 0,
  operator_name text,
  notes text
);

CREATE TABLE ops.application_condition (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  work_execution_id uuid NOT NULL UNIQUE REFERENCES ops.work_execution(id),
  wind_speed_kmh numeric(8,2),
  temperature_c numeric(6,2),
  relative_humidity numeric(6,2),
  thermal_inversion_flag boolean,
  recorded_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE ops.execution_input_consumption (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  work_execution_id uuid NOT NULL REFERENCES ops.work_execution(id),
  input_id uuid NOT NULL REFERENCES agr.input(id),
  qty numeric(14,4) NOT NULL,
  uom_id uuid NOT NULL REFERENCES ref.uom(id)
);

CREATE TABLE ops.field_log_entry (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  plot_season_id uuid NOT NULL REFERENCES core.plot_season(id),
  work_execution_id uuid REFERENCES ops.work_execution(id),
  entry_date date NOT NULL,
  legal_note text NOT NULL,
  certification_tag text
);

CREATE TABLE ops.scouting_event (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  plot_season_id uuid NOT NULL REFERENCES core.plot_season(id),
  observed_at timestamptz NOT NULL,
  category varchar(20) NOT NULL,
  species text,
  severity numeric(6,2),
  action_threshold numeric(6,2),
  geom geometry(Point, 4326),
  CHECK (category IN ('pest','disease','weed','other'))
);

CREATE TABLE ops.irrigation_event (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  plot_season_id uuid NOT NULL REFERENCES core.plot_season(id),
  irrigated_at timestamptz NOT NULL,
  depth_mm numeric(10,2) NOT NULL,
  duration_min integer,
  notes text
);

CREATE TABLE sec.data_access_scope (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  user_id uuid NOT NULL REFERENCES sec.app_user(id),
  campaign_id uuid REFERENCES core.campaign(id),
  establishment_id uuid REFERENCES core.establishment(id),
  plot_id uuid REFERENCES core.plot(id)
);

CREATE TABLE intg.outbox_event (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES core.tenant(id),
  aggregate_type varchar(50) NOT NULL,
  aggregate_id uuid NOT NULL,
  event_type varchar(80) NOT NULL,
  payload jsonb NOT NULL,
  occurred_at timestamptz NOT NULL DEFAULT now(),
  published_at timestamptz
);
